### 1. Stampede-Safe SingleFlight Coalescing (`@kido/resilient-cache`)

When thousands of concurrent requests miss cache on the same hot key simultaneously, SingleFlight collapses them into **1 single backend query**:

<Tabs items={['Python (FastAPI / asyncio)', 'TypeScript (Zero-Dep Slice)']}>
  <Tab value="Python (FastAPI / asyncio)">
    ```python
    import asyncio
    from collections.abc import Awaitable, Callable
    from typing import TypeVar

    T = TypeVar("T")

    class InFlightSingleFlight:
        """Collapses concurrent cache misses on identical keys into one coroutine."""

        def __init__(self) -> None:
            self._inflight: dict[str, asyncio.Future[object]] = {}

        async def dedupe(self, key: str, fetcher: Callable[[], Awaitable[T]]) -> T:
            if key in self._inflight:
                return await asyncio.shield(self._inflight[key])  # type: ignore[return-value]

            loop = asyncio.get_running_loop()
            future: asyncio.Future[object] = loop.create_future()
            self._inflight[key] = future
            try:
                result = await fetcher()
                future.set_result(result)
                return result
            except BaseException as exc:
                future.set_exception(exc)
                raise
            finally:
                self._inflight.pop(key, None)
    ```
  </Tab>
  <Tab value="TypeScript (Zero-Dep Slice)">
    ```typescript
    import { InFlightManager } from '@kido/resilient-cache';

    const flight = new InFlightManager();

    export async function getMarketTicker(symbol: string) {
      return await flight.dedupe(`ticker:${symbol}`, async () => {
        // Only 1 execution hits the database or upstream provider
        return await fetchRealtimeQuotes(symbol);
      });
    }
    ```
  </Tab>
</Tabs>

### 2. Consistent Hash Ring with 150 Virtual Nodes (`@kido/db-sharding`)

Distribute keys uniformly across physical database shards while minimizing key movement during cluster scaling:

<Tabs items={['Python (FastAPI / asyncio)', 'TypeScript (Zero-Dep Slice)']}>
  <Tab value="Python (FastAPI / asyncio)">
    ```python
    from bisect import bisect_right
    import hashlib

    def stable_u32(key: str) -> int:
        return int.from_bytes(hashlib.blake2b(key.encode(), digest_size=4).digest(), "big")

    class ConsistentHashRing:
        def __init__(self, nodes: list[str], vnodes_per_node: int = 150) -> None:
            self._ring: list[tuple[int, str]] = []
            for node in nodes:
                for idx in range(vnodes_per_node):
                    self._ring.append((stable_u32(f"{node}#vn{idx}"), node))
            self._ring.sort(key=lambda item: item[0])
            self._tokens = [token for token, _ in self._ring]

        def get_node(self, key: str) -> str:
            pos = bisect_right(self._tokens, stable_u32(key)) % len(self._ring)
            return self._ring[pos][1]
    ```
  </Tab>
